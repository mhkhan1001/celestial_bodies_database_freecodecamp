CREATE TABLE moon(moon_id SERIAL PRIMARY KEY, name VARCHAR(40) NOT NULL UNIQUE, year_discovered INT, dia_in_km INT, planet_id INT NOT NULL );
CREATE TABLE planet(planet_id SERIAL PRIMARY KEY, name VARCHAR(40) NOT NULL UNIQUE, orital_period_in_years NUMERIC(5,2) NOT NULL, rotation_period_days NUMERIC(5,2), has_moon BOOLEAN, star_id INT NOT NULL);
CREATE TABLE star(star_id SERIAL PRIMARY KEY, name VARCHAR(40) NOT NULL UNIQUE, dist_in_light_years NUMERIC(5,2) NOT NULL, star_brightness NUMERIC(5,2) NOT NULL, approval_date DATE, galaxy_id INT NOT NULL);
CREATE TABLE galaxy(galaxy_id SERIAL PRIMARY KEY, name VARCHAR(40) NOT NULL UNIQUE, meaning TEXT NOT NULL, is_visible_naked_eye BOOLEAN NOT NULL, constellation_id INT NOT NULL);
CREATE TABLE constellation(constellation_id SERIAL PRIMARY KEY, name VARCHAR(40) NOT NULL UNIQUE, year_discovered INT, meaning TEXT NOT NULL, star_id INT NOT NULL UNIQUE);
